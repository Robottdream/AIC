cmake_minimum_required(VERSION 3.8)
project(aic_body_drive)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(aic_body_drive SHARED src/body_drive.cpp)
ament_target_dependencies(aic_body_drive gazebo_ros gazebo_dev rclcpp geometry_msgs nav_msgs sensor_msgs tf2_ros)
target_link_libraries(aic_body_drive ${GAZEBO_LIBRARIES})
install(TARGETS aic_body_drive LIBRARY DESTINATION lib)
ament_package()
