cmake_minimum_required(VERSION 3.8)
project(aic_tf2_fix)
find_package(ament_cmake REQUIRED)
find_package(tf2_ros 0.25.23 EXACT REQUIRED)
find_package(rclcpp REQUIRED)
add_library(aic_tf2_fix SHARED src/async_wait.cpp)
target_compile_features(aic_tf2_fix PUBLIC cxx_std_17)
ament_target_dependencies(aic_tf2_fix tf2_ros rclcpp)
add_executable(tf_async_stress src/stress.cpp)
target_compile_features(tf_async_stress PUBLIC cxx_std_17)
ament_target_dependencies(tf_async_stress tf2_ros rclcpp)
install(TARGETS aic_tf2_fix LIBRARY DESTINATION lib)
install(TARGETS tf_async_stress DESTINATION lib/${PROJECT_NAME})
install(FILES NOTICE DESTINATION share/${PROJECT_NAME})
ament_package()
